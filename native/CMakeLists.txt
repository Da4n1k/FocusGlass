cmake_minimum_required(VERSION 3.22)
project(FocusGlassEffects VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins")
option(BUILD_SHARED_LIBS "Build loadable native plugins" ON)
option(FOCUSGLASS_BUILD_KWIN "Build the version-matched KWin plugin" ON)
if(FOCUSGLASS_BUILD_KWIN)
    find_package(ECM 6.0 REQUIRED NO_MODULE)
    list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
    include(KDEInstallDirs6)
endif()
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core Gui Network)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Gui Network OpenGL)
find_package(PkgConfig REQUIRED)
pkg_check_modules(EPOXY REQUIRED IMPORTED_TARGET epoxy)
add_library(fg_effect_core STATIC effect_core.cpp)
set_target_properties(fg_effect_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(fg_effect_core PUBLIC Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::Gui Qt${QT_VERSION_MAJOR}::Network PkgConfig::EPOXY)
target_compile_options(fg_effect_core PRIVATE -Wall -Wextra -Werror)
find_package(X11 REQUIRED COMPONENTS Xcomposite Xrender Xfixes Xext)
add_executable(focusglass-x11 x11_main.cpp)
target_link_libraries(focusglass-x11 PRIVATE fg_effect_core Qt${QT_VERSION_MAJOR}::OpenGL
    X11::X11 X11::Xcomposite X11::Xrender X11::Xfixes X11::Xext)
target_compile_options(focusglass-x11 PRIVATE -Wall -Wextra -Werror)
add_executable(focusglass-native-test renderer_test.cpp)
target_link_libraries(focusglass-native-test PRIVATE fg_effect_core Qt${QT_VERSION_MAJOR}::OpenGL)
target_compile_options(focusglass-native-test PRIVATE -Wall -Wextra -Werror)
add_executable(focusglass-native-scene scene_test.cpp)
target_link_libraries(focusglass-native-scene PRIVATE fg_effect_core)
if(FOCUSGLASS_BUILD_KWIN)
    if(NOT QT_VERSION_MAJOR EQUAL 6)
        message(FATAL_ERROR "KWin 6 requires Qt 6. Use -DFOCUSGLASS_BUILD_KWIN=OFF for the Qt 5 X11/Fly helper.")
    endif()
    find_package(Qt6 REQUIRED COMPONENTS Widgets DBus)
    find_package(KWin REQUIRED)
    target_link_libraries(focusglass-native-scene PRIVATE KF6::WindowSystem)
    target_compile_definitions(focusglass-native-scene PRIVATE FG_HAVE_KWINDOWSYSTEM)
    kcoreaddons_add_plugin(focusglass SOURCES kwin_effect.cpp
        INSTALL_NAMESPACE "kwin/effects/plugins")
    set_target_properties(focusglass PROPERTIES PREFIX "")
    target_link_libraries(focusglass PRIVATE fg_effect_core KWin::kwin)
    target_compile_options(focusglass PRIVATE -Wall -Wextra -Werror)
endif()
install(TARGETS focusglass-x11 RUNTIME DESTINATION bin)
enable_testing()
add_test(NAME gpu_renderer COMMAND focusglass-native-test)
